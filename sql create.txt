-- LOOKUP TABLES

CREATE TABLE user_types (
    id SERIAL PRIMARY KEY,
    type TEXT NOT NULL
    -- 1 super_admin, 2 admin, 3 standard_user, 4 viewer, 5 blocked
);

CREATE TABLE user_statuses (
    id SERIAL PRIMARY KEY,
    status TEXT NOT NULL
    -- 1 online, 2 idle, 3 do_not_disturb, 4 offline
);

CREATE TABLE account_statuses (
    id SERIAL PRIMARY KEY,
    status TEXT NOT NULL
    -- 1 active, 2 suspended, 3 closed
);

CREATE TABLE themes (
    id SERIAL PRIMARY KEY,
    theme TEXT NOT NULL
    -- 'default', 'rounded', 'sharp', 'gothic', 'times roman'
);

CREATE TABLE relationship_statuses (
    id SERIAL PRIMARY KEY,
    status TEXT NOT NULL
    -- 1 pending, 2 accepted, 3 blocked
);

CREATE TABLE chat_group_permission_types (
    id SERIAL PRIMARY KEY,
    permission_type TEXT NOT NULL
    -- 1 moderator, 2 member, 3 viewer, 4 blocked
);

-- CORE TABLES

CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    username TEXT UNIQUE NOT NULL,
    email TEXT UNIQUE NOT NULL,
    name TEXT NOT NULL,
    password TEXT NOT NULL,
    bio_info TEXT,
    user_type_id INT NOT NULL REFERENCES user_types(id) DEFAULT 4,
    account_status_id INT NOT NULL REFERENCES account_statuses(id) DEFAULT 1,
    status_id INT REFERENCES user_statuses(id) DEFAULT 4,
    is_online BOOLEAN NOT NULL DEFAULT FALSE,
    theme_id INT REFERENCES themes(id) DEFAULT 1,
    theme_dark_mode BOOLEAN NOT NULL DEFAULT FALSE,


    -- Refresh token
    refresh_token TEXT,
    refresh_token_expires_at TIMESTAMPTZ,
    refresh_token_created_at TIMESTAMPTZ,
    refresh_token_updated_at TIMESTAMPTZ,

    -- Audit
    created_by BIGINT REFERENCES users(id),
    updated_by BIGINT REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    -- Light mode colours
    light_theme_primary_colour TEXT NOT NULL DEFAULT '#ffffff',
    light_theme_secondary_colour TEXT NOT NULL DEFAULT '#f0f0f0',
    light_theme_accent_colour TEXT NOT NULL DEFAULT '#0066ff',
    light_theme_sent_colour TEXT NOT NULL DEFAULT '#0066ff',
    light_theme_received_colour TEXT NOT NULL DEFAULT '#e0e0e0',
    light_theme_dark_text_colour TEXT NOT NULL DEFAULT '#000000',
    light_theme_light_text_colour TEXT NOT NULL DEFAULT '#ffffff',

    -- Dark mode colours
    dark_theme_primary_colour TEXT NOT NULL DEFAULT '#1a1a1a',
    dark_theme_secondary_colour TEXT NOT NULL DEFAULT '#2d2d2d',
    dark_theme_accent_colour TEXT NOT NULL DEFAULT '#0066ff',
    dark_theme_sent_colour TEXT NOT NULL DEFAULT '#0066ff',
    dark_theme_received_colour TEXT NOT NULL DEFAULT '#3d3d3d',
    dark_theme_dark_text_colour TEXT NOT NULL DEFAULT '#000000',
    dark_theme_light_text_colour TEXT NOT NULL DEFAULT '#ffffff'
);

CREATE TABLE user_relationships (
    id BIGSERIAL PRIMARY KEY,
    requester_id BIGINT NOT NULL REFERENCES users(id),
    receiver_id BIGINT NOT NULL REFERENCES users(id),
    status_id INT NOT NULL REFERENCES relationship_statuses(id) DEFAULT 1,
    blocked_by BIGINT REFERENCES users(id),
    declined_by BIGINT REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(requester_id, receiver_id)
);

CREATE TABLE chat_groups (
    id BIGSERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    owner_id BIGINT NOT NULL REFERENCES users(id),
    updated_by BIGINT REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE chat_group_permissions (
    id BIGSERIAL PRIMARY KEY,
    group_id BIGINT NOT NULL REFERENCES chat_groups(id),
    user_id BIGINT NOT NULL REFERENCES users(id),
    permission_type_id INT NOT NULL REFERENCES chat_group_permission_types(id),
    updated_by BIGINT REFERENCES users(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(group_id, user_id)
);

CREATE TABLE messages (
    id BIGSERIAL PRIMARY KEY,
    sender_id BIGINT NOT NULL REFERENCES users(id),
    relationship_id BIGINT REFERENCES user_relationships(id),
    group_id BIGINT REFERENCES chat_groups(id),
    message TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT one_target CHECK (
        (relationship_id IS NOT NULL AND group_id IS NULL) OR
        (relationship_id IS NULL AND group_id IS NOT NULL)
    )
);

-- TRIGGER FUNCTIONS

-- MESSAGES

CREATE OR REPLACE FUNCTION notify_message_created()
RETURNS TRIGGER AS $$
DECLARE
    recipient_ids BIGINT[];
BEGIN
    IF NEW.relationship_id IS NOT NULL THEN
        SELECT ARRAY[
            CASE WHEN requester_id = NEW.sender_id THEN receiver_id ELSE requester_id END
        ]
        INTO recipient_ids
        FROM user_relationships
        WHERE id = NEW.relationship_id;
    ELSIF NEW.group_id IS NOT NULL THEN
        SELECT ARRAY_AGG(user_id)
        INTO recipient_ids
        FROM chat_group_permissions
        WHERE group_id = NEW.group_id
        AND permission_type_id != 4
        AND user_id != NEW.sender_id;
    END IF;

    PERFORM pg_notify(
        'message_created',
        json_build_object(
            'message_id', NEW.id,
            'sender_id', NEW.sender_id,
            'relationship_id', NEW.relationship_id,
            'group_id', NEW.group_id,
            'message', NEW.message,
            'created_at', NEW.created_at,
            'recipient_ids', recipient_ids
        )::text
    );
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER message_created_trigger
AFTER INSERT ON messages
FOR EACH ROW EXECUTE FUNCTION notify_message_created();

-- Helper function to get admin ids
CREATE OR REPLACE FUNCTION get_admin_ids()
RETURNS BIGINT[] AS $$
DECLARE
    admin_ids BIGINT[];
BEGIN
    SELECT ARRAY_AGG(id)
    INTO admin_ids
    FROM users
    WHERE user_type_id <= 2
    AND account_status_id = 1;  -- only active admins
    
    RETURN admin_ids;
END;
$$ LANGUAGE plpgsql;
